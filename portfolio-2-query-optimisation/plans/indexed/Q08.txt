-- Q08: Patients with no visit in the last 12 months (indexed)
SELECT p.hospital_no
  FROM patient p
 WHERE p.patient_id NOT IN (SELECT e.patient_id FROM encounter e WHERE e.started_at >= '2025-11-14 00:00+01')
 ORDER BY p.hospital_no;
