-- Q05: Monthly diagnosis counts per branch over the last year (baseline)
SELECT f.facility_code,
       date_trunc('month', e.started_at AT TIME ZONE 'Africa/Lagos')::date AS month,
       d.diagnosis_code,
       count(*) AS cases
  FROM encounter e
  JOIN encounter_diagnosis d USING (encounter_id)
  JOIN facility f USING (facility_id)
 WHERE e.started_at >= '2025-11-01 00:00+01' AND e.started_at < '2026-11-01 00:00+01'
 GROUP BY 1, 2, 3
HAVING count(*) >= 20
 ORDER BY 1, 2, 4 DESC, 3;
